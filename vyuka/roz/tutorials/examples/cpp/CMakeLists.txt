cmake_minimum_required(VERSION 3.16)
project(roz_examples LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(libsvm STATIC third_party/libsvm/svm.cpp)
foreach(program check_pca pca_dbscan classification_pipeline lesson_snippets)
  add_executable(${program} ${program}.cpp)
  if(MSVC)
    target_compile_options(${program} PRIVATE /W4 /utf-8)
  else()
    target_compile_options(${program} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endforeach()
target_link_libraries(classification_pipeline PRIVATE libsvm)
target_link_libraries(lesson_snippets PRIVATE libsvm)
