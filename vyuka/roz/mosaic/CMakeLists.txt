cmake_minimum_required(VERSION 3.16)
project(mosaic_segmenter LANGUAGES CXX)
add_executable(segment segment.cpp)
target_compile_features(segment PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(segment PRIVATE /W4 /utf-8)
else()
  target_compile_options(segment PRIVATE -Wall -Wextra -Wpedantic)
endif()
