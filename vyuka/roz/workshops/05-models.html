Jak K-means přiřadí bod ke středu?How does K-means assign a point to a centre?

Body v grafu jsou v pořadí (1; 1), (1,4; 2), (2; 1), (2; 3), (6; 5), (7; 6) a (8; 5). Počáteční středy jsou C₁ = (1; 1) a C₂ = (2; 3).The seven points in the plot, in order, are (1, 1), (1.4, 2), (2, 1), (2, 3), (6, 5), (7, 6) and (8, 5). The starting centres are C₁ = (1, 1) and C₂ = (2, 3).

Prvním kliknutím na Další krok přiřadíte všechny body ke středům. Po celou dobu přiřazování stojí oba středy na místě. Druhým kliknutím přepočítáte jejich polohy. Porovnáváme druhé mocniny vzdáleností, protože odmocnění by jejich pořadí zachovalo. Při shodě rozhodne nižší číslo středu.The first click on Next step assigns every point to a centre. Both centres stay still while the points are assigned. The second click moves each centre to its new position. We compare squared distances because taking the square root would keep them in the same order. A tie goes to the lower-numbered centre.

středy ← zadané počáteční polohy
předchozí přiřazení ← prázdný seznam
opakuj:
    pro každý bod (x, y):
        nejmenší d² ← ∞
        pro každý střed j v pořadí 1 až K:
            d² ← (x − xⱼ)² + (y − yⱼ)²
            pokud d² < nejmenší d²:
                nejmenší d² ← d²
                přiřaď bod ke shluku j
    pokud je přiřazení všech bodů stejné jako minule: skonči
    ulož současné přiřazení jako předchozí
    pro každý shluk j:
        n ← počet bodů přiřazených k j
        pokud n > 0:
            xⱼ ← součet jejich souřadnic x / n
            yⱼ ← součet jejich souřadnic y / n
        jinak ponech střed na místě
centres ← given starting positions
previous assignments ← empty list
repeat:
    for each point (x, y):
        smallest d² ← ∞
        for each centre j in order from 1 to K:
            d² ← (x − xⱼ)² + (y − yⱼ)²
            if d² < smallest d²:
                smallest d² ← d²
                assign the point to cluster j
    if every point has the same assignment as last time: stop
    save the current assignments as the previous assignments
    for each cluster j:
        n ← number of points assigned to j
        if n > 0:
            xⱼ ← sum of their x coordinates / n
            yⱼ ← sum of their y coordinates / n
        else keep the centre in place

Stejný výpočet v kóduThe same calculation in code

Oba programy číslují středy uvnitř od nuly. Funkce print_step vypíše po každém průchodu štítky od jedničky a polohy středů, stejně jako v tabulce. V balíčku K-means najdete tuto ukázku v souboru kmeans_steps.cpp nebo kmeans_steps.py.Both programs index centres from zero. After each pass, print_step prints labels starting at one and the centre positions, as in the table. The K-means ZIP includes this example as kmeans_steps.cpp or kmeans_steps.py.

Result kmeans(const vector<Point>& points, vector<Point> centres) {
    if (centres.empty()) throw std::invalid_argument("Supply at least one centre");
    vector<size_t> labels;
    for (int iteration = 1; ; ++iteration) {
        vector<size_t> next_labels;
        for (Point p : points) {
            size_t nearest = 0;
            double best = std::numeric_limits<double>::infinity();
            for (size_t j = 0; j < centres.size(); ++j) {
                double dx = p.x - centres[j].x;
                double dy = p.y - centres[j].y;
                double d2 = dx * dx + dy * dy;
                if (d2 < best) {
                    best = d2;
                    nearest = j;
                }
            }
            next_labels.push_back(nearest);
        }
        if (next_labels == labels) {
            print_step(iteration, labels, centres, true);
            return {labels, centres};
        }
        labels = next_labels;

        vector<Point> sums(centres.size(), {0.0, 0.0});
        vector<size_t> counts(centres.size(), 0);
        for (size_t i = 0; i < points.size(); ++i) {
            size_t j = labels[i];
            sums[j].x += points[i].x;
            sums[j].y += points[i].y;
            ++counts[j];
        }
        for (size_t j = 0; j < centres.size(); ++j) {
            if (counts[j] > 0) {
                centres[j] = {sums[j].x / counts[j], sums[j].y / counts[j]};
            }
        }
        print_step(iteration, labels, centres, false);
    }
}
def kmeans(points, centres):
    if not centres:
        raise ValueError("Supply at least one centre")
    centres = list(centres)
    labels = []
    iteration = 1
    while True:
        next_labels = []
        for x, y in points:
            nearest = 0
            best = float("inf")
            for j, (cx, cy) in enumerate(centres):
                d2 = (x - cx) ** 2 + (y - cy) ** 2
                if d2 < best:
                    best = d2
                    nearest = j
            next_labels.append(nearest)
        if next_labels == labels:
            print_step(iteration, labels, centres, True)
            return labels, centres
        labels = next_labels

        sum_x = [0.0] * len(centres)
        sum_y = [0.0] * len(centres)
        counts = [0] * len(centres)
        for (x, y), j in zip(points, labels):
            sum_x[j] += x
            sum_y[j] += y
            counts[j] += 1
        for j, count in enumerate(counts):
            if count > 0:
                centres[j] = (sum_x[j] / count, sum_y[j] / count)
        print_step(iteration, labels, centres, False)
        iteration += 1
